IMPLEMENTATION GUIDE · CHAPTER 6

고급 렌더링 직접 만들기

빛을 더 그럴싸하게가 아니라 물리적으로 맞게 다루는 단계입니다. Cook-Torrance PBR, HDR·톤매핑, 섀도우 맵, 디퍼드 렌더링, SSAO 같은 실시간 기법을 GLSL로, 그리고 오프라인 사실주의의 정점인 레이/패스 트레이서를 CPU C++로 직접 작성합니다. 모든 코드는 개념을 그대로 옮겨 새로 작성한 원본이며, 그대로 컴파일할 수 있는 수준을 지향합니다.

1. 물리 기반 렌더링 (PBR)

퐁/블린-퐁은 "보기 좋게" 조정하는 모델이라 광원·재질을 바꾸면 다시 튜닝해야 합니다. PBR은 에너지 보존과 미세면(microfacet) 이론에 기반해, 재질을 albedo · metallic · roughness 세 값으로 물리적으로 기술합니다. 실시간 표준은 Cook-Torrance BRDF입니다:

f = kd·(albedo/π) + (D·G·F) / (4·(N·V)·(N·L))

  1. D — 법선 분포(GGX/Trowbridge-Reitz): 하프벡터 H 방향으로 정렬된 미세면 비율. roughness가 작을수록 좁고 밝은 하이라이트.
  2. G — 기하 감쇠(Smith + Schlick-GGX): 미세면끼리 서로를 가리거나(masking) 그림자지는(shadowing) 손실.
  3. F — 프레넬(Schlick 근사): 스치는 각도에서 반사율이 커지는 현상. 기본 반사율 F0에서 시작.
  4. 금속-거칠기 워크플로: 비금속은 F0=0.04로 고정, 금속은 albedo가 곧 F0. mix()로 보간.
  5. 에너지 보존: 반사(스페큘러)로 나간 만큼 확산광은 줄인다 → kd = (1-F)*(1-metallic).
pbr.fragglsl
#version 330 core
out vec4 FragColor;

in vec3 WorldPos;   // 월드 좌표
in vec3 Normal;     // 월드 공간 법선
in vec2 TexCoord;

uniform vec3 camPos;

// --- 재질 (텍스처 대신 균일값으로 단순화) ---
uniform vec3  albedo;
uniform float metallic;
uniform float roughness;
uniform float ao;

// --- 광원 (점광원 배열) ---
#define NR_LIGHTS 4
uniform vec3 lightPositions[NR_LIGHTS];
uniform vec3 lightColors[NR_LIGHTS];

const float PI = 3.14159265359;

// D: GGX 법선 분포 함수. H 방향에 미세면이 얼마나 몰려있나
float distributionGGX(vec3 N, vec3 H, float rough) {
    float a  = rough * rough;          // roughness 제곱이 더 지각적으로 선형
    float a2 = a * a;
    float NdotH  = max(dot(N, H), 0.0);
    float NdotH2 = NdotH * NdotH;
    float denom = (NdotH2 * (a2 - 1.0) + 1.0);
    denom = PI * denom * denom;
    return a2 / max(denom, 1e-7);      // 0 나눗셈 방지
}

// Schlick-GGX: 한 방향에 대한 기하 감쇠
float geometrySchlickGGX(float NdotV, float rough) {
    float r = rough + 1.0;
    float k = (r * r) / 8.0;           // 직접광용 k
    return NdotV / (NdotV * (1.0 - k) + k);
}

// Smith: 시선(V)과 광원(L) 양쪽의 가림을 곱한다
float geometrySmith(vec3 N, vec3 V, vec3 L, float rough) {
    float NdotV = max(dot(N, V), 0.0);
    float NdotL = max(dot(N, L), 0.0);
    return geometrySchlickGGX(NdotV, rough) * geometrySchlickGGX(NdotL, rough);
}

// F: 프레넬 Schlick 근사. 스치는 각도일수록 반사율 상승
vec3 fresnelSchlick(float cosTheta, vec3 F0) {
    return F0 + (1.0 - F0) * pow(clamp(1.0 - cosTheta, 0.0, 1.0), 5.0);
}

void main() {
    vec3 N = normalize(Normal);
    vec3 V = normalize(camPos - WorldPos);

    // 비금속 기본 반사율 0.04, 금속이면 albedo 자체가 F0
    vec3 F0 = mix(vec3(0.04), albedo, metallic);

    vec3 Lo = vec3(0.0);               // 나가는 복사휘도 누적
    for (int i = 0; i < NR_LIGHTS; ++i) {
        vec3  L = normalize(lightPositions[i] - WorldPos);
        vec3  H = normalize(V + L);
        float dist = length(lightPositions[i] - WorldPos);
        float atten = 1.0 / (dist * dist);          // 물리적 역제곱 감쇠
        vec3  radiance = lightColors[i] * atten;

        // Cook-Torrance 스페큘러 항
        float D = distributionGGX(N, H, roughness);
        float G = geometrySmith(N, V, L, roughness);
        vec3  F = fresnelSchlick(max(dot(H, V), 0.0), F0);

        vec3  numerator = D * G * F;
        float denom = 4.0 * max(dot(N, V), 0.0) * max(dot(N, L), 0.0) + 1e-4;
        vec3  specular = numerator / denom;

        // 에너지 보존: 반사한 만큼 확산은 줄이고, 금속은 확산이 없다
        vec3 kd = (vec3(1.0) - F) * (1.0 - metallic);

        float NdotL = max(dot(N, L), 0.0);
        Lo += (kd * albedo / PI + specular) * radiance * NdotL;
    }

    // 아주 단순한 앰비언트 (뒤에서 IBL로 대체하는 것이 정석)
    vec3 ambient = vec3(0.03) * albedo * ao;
    vec3 color = ambient + Lo;

    // HDR → LDR: 다음 절의 톤매핑으로 넘기려면 이 두 줄은 빼고 vec4(color,1)만 출력
    color = color / (color + vec3(1.0));          // Reinhard
    color = pow(color, vec3(1.0 / 2.2));          // 감마
    FragColor = vec4(color, 1.0);
}
함정 셰이더 내부의 albedo는 반드시 선형 공간이어야 합니다. sRGB 텍스처를 그대로 쓰면 조명이 두 번 감마 처리되어 어두워집니다. GL_SRGB8 포맷으로 로드하거나 샘플 직후 pow(c, 2.2)로 선형화하세요.
성능 실제 재질감은 IBL(Image-Based Lighting)에서 나옵니다. HDR 환경맵을 미리 필터링해 irradiance 큐브맵(확산)과 prefiltered 큐브맵 + BRDF LUT(스페큘러)을 만들어 앰비언트를 대체하면 추가 조명 없이도 사실적인 결과가 나옵니다.

2. HDR · 톤 매핑

PBR의 밝기는 물리 단위라 1.0을 훌쩍 넘습니다. 기본 8비트 프레임버퍼(0~1)는 이를 잘라내(clamp) 정보를 잃습니다. 해법은 부동소수점 프레임버퍼(GL_RGBA16F)에 HDR로 렌더한 뒤, 마지막에 톤 매핑으로 LDR 범위로 압축하는 것입니다. 여기에 노출(exposure)과 감마를 함께 적용합니다.

  1. 16비트 float 컬러 텍스처를 FBO에 붙이고 씬을 거기에 렌더한다.
  2. 풀스크린 쿼드로 그 텍스처를 샘플, 노출 곱 → 톤매핑 커브 → 감마 순으로 후처리한다.
  3. 블룸을 얹으려면 밝은 부분만 추출 → 가우시안 블러(핑퐁) → 원본에 가산한다.
hdr_fbo.cppcpp
// HDR 렌더 타깃 생성: 16비트 float 컬러 + 깊이 렌더버퍼
GLuint hdrFBO, colorTex, depthRBO;
glGenFramebuffers(1, &hdrFBO);
glBindFramebuffer(GL_FRAMEBUFFER, hdrFBO);

glGenTextures(1, &colorTex);
glBindTexture(GL_TEXTURE_2D, colorTex);
// 핵심: 내부 포맷 GL_RGBA16F → 1.0 초과 값 보존
glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA16F, W, H, 0, GL_RGBA, GL_FLOAT, nullptr);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
glFramebufferTexture2D(GL_FRAMEBUFFER, GL_COLOR_ATTACHMENT0,
                       GL_TEXTURE_2D, colorTex, 0);

glGenRenderbuffers(1, &depthRBO);
glBindRenderbuffer(GL_RENDERBUFFER, depthRBO);
glRenderbufferStorage(GL_RENDERBUFFER, GL_DEPTH_COMPONENT24, W, H);
glFramebufferRenderbuffer(GL_FRAMEBUFFER, GL_DEPTH_ATTACHMENT,
                          GL_RENDERBUFFER, depthRBO);

if (glCheckFramebufferStatus(GL_FRAMEBUFFER) != GL_FRAMEBUFFER_COMPLETE)
    std::fprintf(stderr, "HDR FBO 불완전!\n");
glBindFramebuffer(GL_FRAMEBUFFER, 0);

// --- 렌더 루프 ---
// 1) hdrFBO에 씬 렌더 (톤매핑 없이 HDR 그대로 출력)
glBindFramebuffer(GL_FRAMEBUFFER, hdrFBO);
glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
renderScene();
// 2) 기본 프레임버퍼에 풀스크린 쿼드로 톤매핑
glBindFramebuffer(GL_FRAMEBUFFER, 0);
glClear(GL_COLOR_BUFFER_BIT);
glUseProgram(toneMapProg);
glActiveTexture(GL_TEXTURE0);
glBindTexture(GL_TEXTURE_2D, colorTex);
glUniform1f(glGetUniformLocation(toneMapProg, "exposure"), 1.0f);
drawFullscreenQuad();   // NDC를 덮는 삼각형/쿼드
tonemap.fragglsl
#version 330 core
out vec4 FragColor;
in vec2 uv;

uniform sampler2D hdrBuffer;
uniform float exposure;

// ACES 필믹 근사 (Narkowicz 스타일의 자체 근사)
vec3 acesApprox(vec3 x) {
    const float a = 2.51, b = 0.03, c = 2.43, d = 0.59, e = 0.14;
    return clamp((x * (a * x + b)) / (x * (c * x + d) + e), 0.0, 1.0);
}

void main() {
    vec3 hdr = texture(hdrBuffer, uv).rgb;

    // 1) 노출 적용
    vec3 c = hdr * exposure;

    // 2) 톤매핑 커브 선택 (둘 중 하나)
    // Reinhard: 부드럽지만 하이라이트가 다소 밋밋
    // vec3 mapped = c / (c + vec3(1.0));
    // ACES: 영화적 대비, 하이라이트 롤오프가 자연스럽다
    vec3 mapped = acesApprox(c);

    // 3) 감마 보정 (선형 → sRGB)
    mapped = pow(mapped, vec3(1.0 / 2.2));
    FragColor = vec4(mapped, 1.0);
}
주의 블룸 흐름: (1) 밝기 임계값 초과분만 뽑는 bright-pass, (2) 수평/수직 1D 가우시안을 두 FBO를 번갈아 쓰는 핑퐁 블러로 여러 번, (3) 블러 결과를 HDR 원본에 +로 합산한 뒤 톤매핑. 임계값을 너무 낮추면 화면 전체가 뿌옇게 번집니다.

3. 섀도우 맵 (Shadow Mapping)

그림자의 핵심 아이디어: 광원 시점에서 깊이만 렌더해 "빛이 처음 닿는 거리"를 저장한 뒤, 본 렌더에서 각 픽셀을 광원 공간으로 변환해 그 깊이와 비교합니다. 저장된 값보다 멀면 그림자입니다.

  1. 깊이 전용 FBO를 만든다 (GL_DEPTH_COMPONENT 텍스처, 컬러는 glDrawBuffer(GL_NONE)).
  2. 방향광은 직교 투영으로 light-space 행렬(lightProj * lightView)을 만들어 씬을 그 시점에서 렌더한다.
  3. 본 렌더에서 광원 공간 좌표를 [0,1]로 변환해 깊이맵을 샘플, bias를 더해 비교한다.
  4. 계단 현상을 줄이려 PCF(3×3 평균)로 부드럽게 만든다.
shadow_fbo.cppcpp
const unsigned SHADOW_W = 2048, SHADOW_H = 2048;
GLuint depthFBO, depthMap;
glGenFramebuffers(1, &depthFBO);

glGenTextures(1, &depthMap);
glBindTexture(GL_TEXTURE_2D, depthMap);
glTexImage2D(GL_TEXTURE_2D, 0, GL_DEPTH_COMPONENT24,
             SHADOW_W, SHADOW_H, 0, GL_DEPTH_COMPONENT, GL_FLOAT, nullptr);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_BORDER);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_BORDER);
// 맵 밖은 "그림자 아님(깊이 1.0)"으로 처리
float border[] = { 1.0f, 1.0f, 1.0f, 1.0f };
glTexParameterfv(GL_TEXTURE_2D, GL_TEXTURE_BORDER_COLOR, border);

glBindFramebuffer(GL_FRAMEBUFFER, depthFBO);
glFramebufferTexture2D(GL_FRAMEBUFFER, GL_DEPTH_ATTACHMENT,
                       GL_TEXTURE_2D, depthMap, 0);
glDrawBuffer(GL_NONE);   // 컬러 출력 없음
glReadBuffer(GL_NONE);
glBindFramebuffer(GL_FRAMEBUFFER, 0);

// --- 1패스: 광원 시점 깊이 렌더 ---
glm::mat4 lightProj = glm::ortho(-10.f, 10.f, -10.f, 10.f, 1.0f, 25.f);
glm::mat4 lightView = glm::lookAt(lightPos, glm::vec3(0.0f), glm::vec3(0,1,0));
glm::mat4 lightSpace = lightProj * lightView;

glViewport(0, 0, SHADOW_W, SHADOW_H);
glBindFramebuffer(GL_FRAMEBUFFER, depthFBO);
glClear(GL_DEPTH_BUFFER_BIT);
glCullFace(GL_FRONT);           // 앞면 컬링 → peter-panning 완화 트릭
renderSceneDepthOnly(lightSpace);
glCullFace(GL_BACK);
glBindFramebuffer(GL_FRAMEBUFFER, 0);
// --- 2패스: 본 렌더 (lightSpace, depthMap을 셰이더로 넘김) ---
lighting_shadow.frag (그림자 부분)glsl
uniform sampler2D shadowMap;
in vec4 FragPosLightSpace;   // lightSpace * worldPos (버텍스에서 계산)

// 0.0 = 완전히 빛, 1.0 = 완전한 그림자
float shadowCalc(vec4 fragPosLS, vec3 N, vec3 L) {
    // 원근 분할 후 [-1,1] → [0,1] 텍스처 좌표로
    vec3 proj = fragPosLS.xyz / fragPosLS.w;
    proj = proj * 0.5 + 0.5;
    if (proj.z > 1.0) return 0.0;       // 절두체 밖은 그림자 아님

    float current = proj.z;             // 이 픽셀의 광원 기준 깊이

    // 경사에 따라 bias를 키워 shadow acne 억제
    float bias = max(0.05 * (1.0 - dot(N, L)), 0.005);

    // PCF 3x3: 주변 9개 텍셀과 비교해 평균
    float shadow = 0.0;
    vec2 texel = 1.0 / vec2(textureSize(shadowMap, 0));
    for (int x = -1; x <= 1; ++x)
        for (int y = -1; y <= 1; ++y) {
            float pcf = texture(shadowMap, proj.xy + vec2(x, y) * texel).r;
            shadow += (current - bias > pcf) ? 1.0 : 0.0;
        }
    return shadow / 9.0;
}
// 사용: vec3 color = ambient + (1.0 - shadowCalc(...)) * (diffuse + specular);
함정 Shadow acne(줄무늬 자기그림자)는 bias가 부족할 때, Peter-panning(물체가 그림자에서 떠 보임)은 bias가 과할 때 생깁니다. 경사 기반 bias + 깊이 패스에서 앞면 컬링(GL_FRONT)을 함께 쓰면 균형이 좋습니다. 근/원 평면을 씬에 딱 맞게 좁히면 깊이 정밀도가 올라가 acne가 줄어듭니다.

4. 디퍼드 렌더링 (Deferred Rendering)

포워드 렌더링은 픽셀마다 모든 광원을 계산합니다 — 광원이 많아지면 가려져 버려질 픽셀까지 셰이딩하느라 낭비가 큽니다. 디퍼드는 두 단계로 나눕니다: (1) 지오메트리 패스에서 위치·법선·알베도를 여러 텍스처(G-buffer)에 한 번에 기록하고, (2) 라이팅 패스에서 화면의 보이는 픽셀에 대해서만 G-buffer를 읽어 조명을 계산합니다. 조명 비용이 광원×화면픽셀로 고정됩니다.

  1. 3개의 컬러 어태치먼트(position, normal, albedo+spec)를 glDrawBuffers로 MRT 설정한다.
  2. 지오메트리 패스 셰이더가 그 3개에 각각 기록한다.
  3. 라이팅 패스는 풀스크린 쿼드로 G-buffer를 샘플, 광원 루프를 돈다.
gbuffer.cppcpp
GLuint gBuffer, gPosition, gNormal, gAlbedoSpec, gDepth;
glGenFramebuffers(1, &gBuffer);
glBindFramebuffer(GL_FRAMEBUFFER, gBuffer);

auto makeTex = [&](GLuint& tex, GLint internal, GLenum fmt, GLenum type, int slot){
    glGenTextures(1, &tex);
    glBindTexture(GL_TEXTURE_2D, tex);
    glTexImage2D(GL_TEXTURE_2D, 0, internal, W, H, 0, fmt, type, nullptr);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST);
    glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST);
    glFramebufferTexture2D(GL_FRAMEBUFFER, GL_COLOR_ATTACHMENT0 + slot,
                           GL_TEXTURE_2D, tex, 0);
};
makeTex(gPosition,   GL_RGBA16F, GL_RGBA, GL_FLOAT,         0); // 뷰공간 위치
makeTex(gNormal,     GL_RGBA16F, GL_RGBA, GL_FLOAT,         1); // 뷰공간 법선
makeTex(gAlbedoSpec, GL_RGBA8,   GL_RGBA, GL_UNSIGNED_BYTE, 2); // rgb=albedo, a=spec

// 이 3개에 동시에 출력하겠다고 선언 (MRT)
GLuint attachments[3] = { GL_COLOR_ATTACHMENT0,
                          GL_COLOR_ATTACHMENT1,
                          GL_COLOR_ATTACHMENT2 };
glDrawBuffers(3, attachments);

glGenRenderbuffers(1, &gDepth);
glBindRenderbuffer(GL_RENDERBUFFER, gDepth);
glRenderbufferStorage(GL_RENDERBUFFER, GL_DEPTH_COMPONENT24, W, H);
glFramebufferRenderbuffer(GL_FRAMEBUFFER, GL_DEPTH_ATTACHMENT,
                          GL_RENDERBUFFER, gDepth);
glBindFramebuffer(GL_FRAMEBUFFER, 0);
geometry.fragglsl
#version 330 core
layout (location = 0) out vec3 gPosition;    // 어태치먼트 0
layout (location = 1) out vec3 gNormal;      // 어태치먼트 1
layout (location = 2) out vec4 gAlbedoSpec;  // 어태치먼트 2

in vec3 FragPos;   // 뷰 공간 위치
in vec3 Normal;    // 뷰 공간 법선
in vec2 TexCoord;

uniform sampler2D diffuseTex;
uniform float specularStrength;

void main() {
    gPosition = FragPos;
    gNormal   = normalize(Normal);
    gAlbedoSpec.rgb = texture(diffuseTex, TexCoord).rgb;
    gAlbedoSpec.a   = specularStrength;   // 알파 채널에 스페큘러 세기 저장
}
lighting_pass.fragglsl
#version 330 core
out vec4 FragColor;
in vec2 uv;

uniform sampler2D gPosition;
uniform sampler2D gNormal;
uniform sampler2D gAlbedoSpec;

struct Light { vec3 Position; vec3 Color; float Linear; float Quadratic; };
#define NR_LIGHTS 64
uniform Light lights[NR_LIGHTS];

void main() {
    // G-buffer에서 픽셀 속성 복원
    vec3  FragPos = texture(gPosition, uv).rgb;
    vec3  N       = normalize(texture(gNormal, uv).rgb);
    vec3  Albedo  = texture(gAlbedoSpec, uv).rgb;
    float Spec    = texture(gAlbedoSpec, uv).a;

    vec3 V = normalize(-FragPos);        // 뷰공간에서 카메라는 원점
    vec3 lighting = Albedo * 0.1;        // 앰비언트

    for (int i = 0; i < NR_LIGHTS; ++i) {
        vec3  L = normalize(lights[i].Position - FragPos);
        float diff = max(dot(N, L), 0.0);
        vec3  H = normalize(L + V);
        float spec = pow(max(dot(N, H), 0.0), 32.0) * Spec;

        float d = length(lights[i].Position - FragPos);
        float atten = 1.0 / (1.0 + lights[i].Linear * d
                                 + lights[i].Quadratic * d * d);
        lighting += (diff * Albedo + spec) * lights[i].Color * atten;
    }
    FragColor = vec4(lighting, 1.0);
}
성능 대역폭 트레이드오프: 디퍼드는 픽셀당 여러 개의 넓은 G-buffer를 쓰고 읽어 메모리 대역폭을 많이 씁니다. 그 대신 오버드로우 셰이딩이 사라져 광원이 많을 때 압승입니다. 단점은 (1) MSAA가 어렵고 (2) 반투명 처리가 안 돼(포워드로 별도 패스 필요), (3) 재질 다양성이 G-buffer 크기에 제한됩니다.

5. 스크린 스페이스 효과 (SSAO 포함)

화면에 이미 그려진 정보(깊이·법선·컬러)만으로 후처리하는 기법군입니다. 대표가 SSAO(Screen-Space Ambient Occlusion): 각 픽셀 주변 반구에서 뷰 공간 샘플을 뿌리고, 그 샘플이 실제 깊이보다 뒤에 가려져 있으면 가림(occlusion)으로 세어 틈새를 어둡게 만듭니다. 디퍼드의 G-buffer와 궁합이 좋습니다.

  1. 법선 방향 반구 안에 커널 샘플 벡터들을 생성한다(중심에 몰리게 가중).
  2. 4×4 랜덤 회전 노이즈 텍스처로 커널을 화면 타일 단위로 돌려 밴딩을 없앤다.
  3. SSAO 셰이더가 커널을 뷰 공간에 놓고 깊이 비교 + 거리 범위 체크로 가림을 누적한다.
  4. 노이즈 때문에 지저분하므로 4×4 박스 블러로 마무리한다.
ssao_setup.cppcpp
#include <random>
#include <vector>
#include <glm/glm.hpp>

std::uniform_real_distribution<float> rnd(0.0f, 1.0f);
std::default_random_engine gen;
auto lerp = [](float a, float b, float t){ return a + t * (b - a); };

// 1) 반구 커널: 법선(+z) 반구 안의 샘플 64개
std::vector<glm::vec3> kernel;
for (unsigned i = 0; i < 64; ++i) {
    glm::vec3 s(rnd(gen) * 2.0f - 1.0f,   // x: -1..1
               rnd(gen) * 2.0f - 1.0f,   // y: -1..1
               rnd(gen));                // z:  0..1 → 반구
    s = glm::normalize(s) * rnd(gen);
    // 중심에 샘플을 더 몰아준다 (가까운 가림이 중요)
    float scale = (float)i / 64.0f;
    s *= lerp(0.1f, 1.0f, scale * scale);
    kernel.push_back(s);
}

// 2) 4x4 랜덤 회전 노이즈 (z=0인 접평면 벡터)
std::vector<glm::vec3> noise;
for (unsigned i = 0; i < 16; ++i)
    noise.emplace_back(rnd(gen) * 2.0f - 1.0f, rnd(gen) * 2.0f - 1.0f, 0.0f);

GLuint noiseTex;
glGenTextures(1, &noiseTex);
glBindTexture(GL_TEXTURE_2D, noiseTex);
glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA16F, 4, 4, 0, GL_RGB, GL_FLOAT, noise.data());
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_REPEAT);  // 화면에 타일링
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_REPEAT);
ssao.fragglsl
#version 330 core
out float FragColor;   // 가림 값 하나만 출력 (R 채널)
in vec2 uv;

uniform sampler2D gPosition;  // 뷰공간 위치
uniform sampler2D gNormal;    // 뷰공간 법선
uniform sampler2D texNoise;   // 4x4 회전 노이즈

uniform vec3 samples[64];
uniform mat4 projection;      // 뷰 → 클립

const int   KERNEL = 64;
const float radius = 0.5;
const float bias   = 0.025;

void main() {
    vec2 noiseScale = vec2(textureSize(gPosition, 0)) / 4.0; // 노이즈 타일 스케일

    vec3 fragPos = texture(gPosition, uv).xyz;
    vec3 normal  = normalize(texture(gNormal, uv).xyz);
    vec3 randVec = normalize(texture(texNoise, uv * noiseScale).xyz);

    // Gram-Schmidt로 픽셀별 회전 TBN 기저 구성
    vec3 tangent   = normalize(randVec - normal * dot(randVec, normal));
    vec3 bitangent = cross(normal, tangent);
    mat3 TBN       = mat3(tangent, bitangent, normal);

    float occlusion = 0.0;
    for (int i = 0; i < KERNEL; ++i) {
        // 커널 샘플을 뷰 공간의 픽셀 주변으로 이동
        vec3 samplePos = fragPos + TBN * samples[i] * radius;

        // 그 샘플을 화면 좌표로 투영해 저장된 깊이를 얻는다
        vec4 offset = projection * vec4(samplePos, 1.0);
        offset.xyz /= offset.w;
        offset.xyz = offset.xyz * 0.5 + 0.5;

        float sampleDepth = texture(gPosition, offset.xy).z;

        // 범위 체크: 깊이 차가 반지름을 넘으면 무시(경계 오검출 방지)
        float rangeCheck = smoothstep(0.0, 1.0, radius / abs(fragPos.z - sampleDepth));
        // 저장 깊이가 샘플보다 카메라에 가까우면 가려진 것
        occlusion += (sampleDepth >= samplePos.z + bias ? 1.0 : 0.0) * rangeCheck;
    }
    FragColor = 1.0 - (occlusion / float(KERNEL));  // 1=열림, 0=완전 가림
}

6. 레이 트레이싱 (CPU)

래스터화가 "삼각형을 화면에 투영"한다면, 레이 트레이싱은 반대로 픽셀마다 광선을 쏘아 무엇과 먼저 부딪히는지 찾습니다. 그림자·반사·굴절이 자연스럽게 나옵니다 — 그림자 광선을 광원으로 쏘고, 반사 광선을 재귀적으로 추적하면 됩니다. 외부 의존성 없는 자립 실행 C++ 예제입니다.

  1. Ray(원점+방향), Sphere, 재질을 정의한다.
  2. 레이-구 교차(2차 방정식)를 푼다.
  3. 가장 가까운 교차점에서 램버트 셰이딩 + 그림자 광선으로 가림을 판정한다.
  4. 거울 재질이면 반사 방향으로 재귀 추적(최대 N회 바운스).
  5. 결과를 PPM 이미지로 저장한다.
raytracer.cppcpp
// g++ -O2 raytracer.cpp -o rt && ./rt > out.ppm
#include <cmath>
#include <cstdio>
#include <vector>
#include <limits>
#include <algorithm>

// --- 최소 3D 벡터 ---
struct Vec3 {
    float x, y, z;
    Vec3(float a=0, float b=0, float c=0):x(a),y(b),z(c){}
    Vec3 operator+(const Vec3& o) const { return {x+o.x, y+o.y, z+o.z}; }
    Vec3 operator-(const Vec3& o) const { return {x-o.x, y-o.y, z-o.z}; }
    Vec3 operator*(float s)      const { return {x*s, y*s, z*s}; }
    Vec3 operator*(const Vec3& o) const { return {x*o.x, y*o.y, z*o.z}; }
};
float dot(const Vec3& a, const Vec3& b){ return a.x*b.x + a.y*b.y + a.z*b.z; }
float len(const Vec3& a){ return std::sqrt(dot(a,a)); }
Vec3 normalize(const Vec3& a){ float l = len(a); return a * (1.0f/l); }
Vec3 reflect(const Vec3& d, const Vec3& n){ return d - n * (2.0f * dot(d,n)); }

struct Ray { Vec3 o, d; };            // 원점, 정규화된 방향

struct Sphere {
    Vec3  center;
    float radius;
    Vec3  albedo;
    float reflectivity;               // 0=난반사, 1=완전 거울
};

// 레이-구 교차: |o + t d - c|^2 = r^2 를 t에 대해 푼다
bool hitSphere(const Sphere& s, const Ray& r, float& t) {
    Vec3 oc = r.o - s.center;
    float b = 2.0f * dot(oc, r.d);
    float c = dot(oc, oc) - s.radius * s.radius;
    float disc = b*b - 4.0f*c;        // a=1 (d가 단위벡터)
    if (disc < 0.0f) return false;
    float sq = std::sqrt(disc);
    float t0 = (-b - sq) * 0.5f;
    float t1 = (-b + sq) * 0.5f;
    t = (t0 > 1e-3f) ? t0 : t1;        // 앞쪽 교차 우선
    return t > 1e-3f;
}

std::vector<Sphere> scene = {
    {{ 0.0f, -0.5f, -4.0f}, 1.0f, {0.9f, 0.3f, 0.3f}, 0.3f}, // 빨강, 약한 반사
    {{ 2.2f,  0.0f, -6.0f}, 1.2f, {0.3f, 0.6f, 0.9f}, 0.8f}, // 파랑, 거울성
    {{-2.2f,  0.0f, -5.0f}, 1.0f, {0.3f, 0.9f, 0.4f}, 0.0f}, // 초록, 무광
    {{ 0.0f,-101.0f,-5.0f}, 100.0f,{0.7f, 0.7f, 0.7f}, 0.1f} // 바닥(큰 구)
};
Vec3 lightPos(5.0f, 8.0f, 0.0f);

// 씬에서 가장 가까운 교차 구를 찾는다
int closest(const Ray& r, float& tHit) {
    int idx = -1; tHit = std::numeric_limits<float>::max();
    for (int i = 0; i < (int)scene.size(); ++i) {
        float t;
        if (hitSphere(scene[i], r, t) && t < tHit) { tHit = t; idx = i; }
    }
    return idx;
}

// 재귀적 광선 추적
Vec3 trace(const Ray& r, int depth) {
    if (depth <= 0) return {0,0,0};

    float t; int i = closest(r, t);
    if (i < 0) return {0.5f, 0.7f, 1.0f};      // 하늘색 배경

    const Sphere& s = scene[i];
    Vec3 p = r.o + r.d * t;                     // 교차점
    Vec3 n = normalize(p - s.center);           // 표면 법선
    Vec3 toL = normalize(lightPos - p);

    // 그림자 광선: 교차점 → 광원 사이에 막히는 물체가 있나
    Ray shadow{ p + n * 1e-3f, toL };
    float st; bool inShadow = (closest(shadow, st) >= 0);

    float diff = inShadow ? 0.0f : std::max(0.0f, dot(n, toL));
    Vec3 local = s.albedo * (0.1f + 0.9f * diff);   // 앰비언트 + 램버트

    // 거울 반사 재귀
    if (s.reflectivity > 0.0f) {
        Ray refl{ p + n * 1e-3f, normalize(reflect(r.d, n)) };
        Vec3 rc = trace(refl, depth - 1);
        local = local * (1.0f - s.reflectivity) + rc * s.reflectivity;
    }
    return local;
}

int main() {
    const int W = 640, H = 480, MAX_DEPTH = 5;
    const float aspect = (float)W / H;
    std::printf("P3\n%d %d\n255\n", W, H);      // PPM 헤더

    Vec3 cam(0, 0, 0);
    for (int y = 0; y < H; ++y) {
        for (int x = 0; x < W; ++x) {
            // 픽셀 → NDC → 카메라 광선 (fov 90도 가정, 초점거리 1)
            float px = (2.0f * (x + 0.5f) / W - 1.0f) * aspect;
            float py = 1.0f - 2.0f * (y + 0.5f) / H;
            Ray r{ cam, normalize(Vec3(px, py, -1.0f)) };

            Vec3 col = trace(r, MAX_DEPTH);
            auto to255 = [](float v){
                return (int)(std::clamp(std::pow(v, 1.0f/2.2f), 0.0f, 1.0f) * 255);
            };
            std::printf("%d %d %d ", to255(col.x), to255(col.y), to255(col.z));
        }
    }
    return 0;
}
함정 그림자·반사 광선의 원점을 표면에서 n * 1e-3f만큼 살짝 띄우지 않으면, 부동소수점 오차로 자기 자신과 즉시 교차해 검은 얼룩(shadow acne)이 생깁니다. 이 epsilon 오프셋은 필수입니다.

7. 패스 트레이싱 (CPU)

레이 트레이싱은 직접광과 완벽한 반사만 다뤘습니다. 패스 트레이싱은 렌더링 방정식을 몬테카를로로 풉니다 — 각 표면에서 무작위 방향으로 광선을 튕겨 빛의 경로를 확률적으로 샘플링하고, 픽셀마다 수많은 경로를 평균냅니다. 그러면 부드러운 그림자·간접광·색번짐(color bleeding)이 공짜로 나옵니다. 대가는 노이즈이며, 샘플 수를 늘려야 사라집니다.

  1. 표면에서 코사인 가중 반구 샘플링으로 다음 방향을 뽑는다(램버트 BRDF와 상쇄되어 식이 단순해짐).
  2. 빛에 부딪히면 그 방출값을 경로 기여로 반환, 아니면 재귀로 계속 튕긴다.
  3. 픽셀당 SPP(samples per pixel)만큼 반복해 평균 → 톤매핑.

// g++ -O2 pathtracer.cpp -o pt && ./pt > cornell.ppm

#include <cstdio>
#include <vector>

#include <algorithm>

    Vec3(float a=0,float b=0,float c=0):x(a),y(b),z(c){}
    Vec3 operator+(const Vec3&o)const{return{x+o.x,y+o.y,z+o.z};}
    Vec3 operator-(const Vec3&o)const{return{x-o.x,y-o.y,z-o.z};}

    Vec3 operator*(const Vec3&o)const{return{x*o.x,y*o.y,z*o.z};}

float dot(const Vec3&a,const Vec3&b){return a.x*b.x+a.y*b.y+a.z*b.z;}
Vec3 normalize(const Vec3&a){float l=std::sqrt(dot(a,a));return a*(1.0f/l);}
Vec3 cross(const Vec3&a,const Vec3&b){
    return {a.y*b.z-a.z*b.y, a.z*b.x-a.x*b.z, a.x*b.y-a.y*b.x};
}

struct Ray { Vec3 o, d; };
// emission이 0이 아니면 광원
struct Sphere { Vec3 c; float r; Vec3 albedo; Vec3 emission; };

// 코넬 박스: 큰 구들로 벽을 만들고, 위에 밝은 구 하나를 광원으로
std::vector<Sphere> scene = {
    {{-1e5f-2, 0, -5}, 1e5f, {0.75f,0.25f,0.25f}, {0,0,0}}, // 왼벽 빨강
    {{ 1e5f+2, 0, -5}, 1e5f, {0.25f,0.35f,0.75f}, {0,0,0}}, // 오른벽 파랑
    {{0, 0, -1e5f-8}, 1e5f, {0.75f,0.75f,0.75f}, {0,0,0}},  // 뒷벽 흰색
    {{0, -1e5f-2, -5}, 1e5f, {0.75f,0.75f,0.75f}, {0,0,0}}, // 바닥
    {{0,  1e5f+3, -5}, 1e5f, {0.75f,0.75f,0.75f}, {0,0,0}}, // 천장
    {{-0.9f,-1.3f,-5.5f}, 0.7f, {0.9f,0.9f,0.9f}, {0,0,0}}, // 물체 구 1
    {{ 0.9f,-1.3f,-4.2f}, 0.7f, {0.9f,0.9f,0.9f}, {0,0,0}}, // 물체 구 2
    {{0, 2.6f, -5}, 0.6f, {0,0,0}, {12,12,12}}              // 천장 광원
};

std::mt19937 rng(1234);
std::uniform_real_distribution<float> U(0.0f, 1.0f);

bool hit(const Sphere&s, const Ray&r, float&t){
    Vec3 oc = r.o - s.c;
    float b = dot(oc, r.d);
    float c = dot(oc,oc) - s.r*s.r;
    float disc = b*b - c;
    if (disc < 0) return false;
    float sq = std::sqrt(disc);
    t = -b - sq; if (t > 1e-3f) return true;
    t = -b + sq; return t > 1e-3f;
}
int closest(const Ray&r, float&tHit){
    int idx=-1; tHit=std::numeric_limits<float>::max();
    for (int i=0;i<(int)scene.size();++i){
        float t; if (hit(scene[i],r,t) && t<tHit){tHit=t; idx=i;}
    }
    return idx;
}

// 법선 n 주변 코사인 가중 반구 샘플
Vec3 cosineHemisphere(const Vec3& n){
    float u1 = U(rng), u2 = U(rng);
    float r = std::sqrt(u1);
    float theta = 6.2831853f * u2;
    // n을 z축으로 하는 직교 기저 구성
    Vec3 a = (std::fabs(n.x) > 0.1f) ? Vec3(0,1,0) : Vec3(1,0,0);
    Vec3 t = normalize(cross(a, n));
    Vec3 b = cross(n, t);
    Vec3 local(r*std::cos(theta), r*std::sin(theta), std::sqrt(1.0f-u1));
    return normalize(t*local.x + b*local.y + n*local.z);
}

// 경로 추적: 반환값은 이 경로가 카메라로 가져오는 복사휘도
Vec3 radiance(Ray r, int depth){
    if (depth <= 0) return {0,0,0};
    float t; int i = closest(r, t);
    if (i < 0) return {0,0,0};                  // 허공 = 검정

    const Sphere& s = scene[i];
    Vec3 p = r.o + r.d * t;
    Vec3 n = normalize(p - s.c);
    if (dot(n, r.d) > 0) n = n * -1.0f;          // 법선을 시선 쪽으로

    // 광원에 닿으면 방출값 반환 (경로 종료)
    if (s.emission.x + s.emission.y + s.emission.z > 0.0f)
        return s.emission;

    // 러시안 룰렛으로 무한 재귀 방지 (밝은 표면일수록 오래 살린다)
    float pSurvive = std::max({s.albedo.x, s.albedo.y, s.albedo.z});
    if (depth < 4) {
        if (U(rng) > pSurvive) return {0,0,0};
    } else pSurvive = 1.0f;

    // 코사인 샘플링에서는 (albedo/π)·cos·(π/cos) = albedo 만 남는다
    Vec3 newDir = cosineHemisphere(n);
    Ray  bounce{ p + n * 1e-3f, newDir };
    Vec3 incoming = radiance(bounce, depth - 1);
    return s.albedo * incoming * (1.0f / pSurvive);
}

int main(){
    const int W=400, H=400, SPP=256, MAX_DEPTH=8;
    std::printf("P3\n%d %d\n255\n", W, H);
    Vec3 cam(0, 0, 0);

    for (int y=0; y<H; ++y){
        for (int x=0; x<W; ++x){
            Vec3 acc(0,0,0);
            for (int s=0; s<SPP; ++s){
                // 픽셀 내 지터링으로 안티앨리어싱까지 겸함
                float px = (2.0f*(x + U(rng))/W - 1.0f);
                float py = 1.0f - 2.0f*(y + U(rng))/H;
                Ray r{ cam, normalize(Vec3(px, py, -1.0f)) };
                acc = acc + radiance(r, MAX_DEPTH);
            }
            acc = acc * (1.0f / SPP);            // 평균
            // Reinhard 톤매핑 + 감마
            auto tm = [](float v){
                v = v / (1.0f + v);
                return (int)(std::pow(std::clamp(v,0.0f,1.0f), 1.0f/2.2f) * 255);
            };
            std::printf("%d %d %d ", tm(acc.x), tm(acc.y), tm(acc.z));
        }
    }
    return 0;
}
성능 노이즈는 표준편차가 1/√N로 줄어듭니다 — 노이즈를 절반으로 낮추려면 샘플을 4배 써야 합니다. 그래서 실무는 중요도 샘플링(광원 방향 직접 샘플링), 다중 중요도 샘플링(MIS), 그리고 마지막에 디노이저(OIDN/OptiX)로 적은 SPP를 보정합니다.
함정 코사인 가중 샘플링을 쓰면 PDF가 cos/π라 램버트 BRDF의 albedo/π·cos와 상쇄되어 남는 항이 albedo뿐입니다. 균등(uniform) 샘플링을 쓰면서 이 상쇄를 잊고 그대로 곱하면 결과가 너무 어둡거나 밝아집니다.

8. 전역 조명 (Global Illumination)

전역 조명(GI)은 빛이 표면끼리 여러 번 튕기며 만드는 간접광입니다 — 앞 절의 패스 트레이서는 이미 완전한 GI를 계산합니다(재귀 바운스가 곧 간접광이고, 빨간 벽 옆 흰 구가 붉게 물드는 color bleeding이 그 증거). 즉 정답은 이미 있습니다. 문제는 실시간이라는 제약입니다.

실시간에서는 패스 트레이싱을 통째로 돌릴 수 없으니, GI를 근사하는 여러 기법이 발전했습니다:

기법아이디어장점한계
라이트맵 (베이킹) 오프라인에서 GI를 계산해 텍스처에 굽는다 런타임 비용 거의 0, 품질 최고 정적 씬 전용, 굽는 시간 김
Irradiance Probes 공간에 프로브를 배치해 방향별 간접광 저장, 동적 물체가 보간해 샘플 동적 물체 대응, 비교적 저렴 프로브 밀도·배치에 품질 의존, 누수(light leak)
Voxel GI (VXGI) 씬을 복셀화해 원뿔 추적(cone tracing)으로 간접광 근사 완전 동적 GI 가능 메모리·성능 부담, 복셀 해상도 한계
SSGI SSAO처럼 화면 공간에서 간접 반사광을 근사 저렴, 기존 G-buffer 재사용 화면 밖 정보는 못 씀(SSR과 같은 한계)
하드웨어 RT / RTGI RT 코어로 소수 광선을 실시간 추적 + 강력한 디노이징 동적·정확, 반사/그림자 통합 고성능 GPU 필요, 여전히 노이즈/시간적 안정성 과제
주의 최신 흐름은 하이브리드입니다: 래스터화로 기본 화면을 그리고(빠름), 반사·그림자·GI 일부만 하드웨어 RT로 소량 추적한 뒤 시공간(spatio-temporal) 디노이저로 다듬습니다. UE5의 Lumen이 대표적으로 SDF/화면공간 추적과 하드웨어 RT를 상황에 따라 섞습니다.

요약하면 — 오프라인은 패스 트레이싱으로 정답을, 실시간은 근사와 하드웨어 RT의 조합으로 "충분히 그럴싸한 정답"을 추구하는 방향으로 수렴하고 있습니다.
