IMPLEMENTATION GUIDE · CHAPTER 7

실전 렌더링 기법

조명과 그림자가 있어도 화면이 "게임 같지 않다"면 대개 원인은 이 장의 기법들에 있습니다. 계단 현상을 없애는 안티앨리어싱, 색이 정확해지는 감마 · 선형 색공간, 멀리 있는 텍스처의 지글거림을 잡는 밉맵, 유리·연기의 투명 처리, 공간감을 더하는 앰비언트 오클루전, 그리고 대규모 씬을 굴러가게 하는 LOD · 컬링까지 — 모두 복사해 바로 컴파일할 수 있는 수준의 모던 OpenGL 3.3+ 예제로 다룹니다.

1. 안티앨리어싱 (Anti-aliasing)

기하가 픽셀 격자에 스냅되면서 생기는 계단 현상(jaggies)을 부드럽게 만드는 기법입니다. OpenGL에서 가장 흔한 세 갈래는 MSAA(하드웨어 멀티샘플), FXAA(후처리 셰이더), TAA(시간 누적)입니다. 목적과 비용이 다르므로 순서대로 봅니다.

1-1. MSAA — 가장 쉬운 길 (기본 프레임버퍼)

후처리 파이프라인이 없다면, 창을 만들 때 힌트 한 줄이면 끝납니다. GLFW가 멀티샘플 백버퍼를 만들어 주고, 우리는 활성화만 하면 됩니다.

  1. 컨텍스트 생성 전에 샘플 수를 힌트로 요청한다.
  2. GLAD 로드 후 GL_MULTISAMPLE을 켠다.
  3. 평소처럼 그린다 — 하드웨어가 알아서 엣지를 다중 샘플링한다.
msaa_default.cppcpp
// ── 창 생성 전: 4x MSAA 백버퍼 요청 ──
glfwWindowHint(GLFW_CONTEXT_VERSION_MAJOR, 3);
glfwWindowHint(GLFW_CONTEXT_VERSION_MINOR, 3);
glfwWindowHint(GLFW_OPENGL_PROFILE, GLFW_OPENGL_CORE_PROFILE);
glfwWindowHint(GLFW_SAMPLES, 4);          // 픽셀당 4 샘플

GLFWwindow* window = glfwCreateWindow(1280, 720, "MSAA", nullptr, nullptr);
glfwMakeContextCurrent(window);
gladLoadGLLoader((GLADloadproc)glfwGetProcAddress);

glEnable(GL_MULTISAMPLE);                  // 멀티샘플 래스터화 활성화
glEnable(GL_DEPTH_TEST);

// 렌더 루프는 평소와 동일 — 별도 코드 없이 엣지가 부드러워진다.
while (!glfwWindowShouldClose(window)) {
    glClearColor(0.06f, 0.07f, 0.10f, 1.0f);
    glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
    drawScene();
    glfwSwapBuffers(window);
    glfwPollEvents();
}
성능 MSAA는 지오메트리 엣지에만 효과가 있고, 셰이더는 픽셀당 한 번만 돕니다 (샘플당이 아님). 그래서 비용 대비 화질이 좋지만, 알파 텍스처의 잘린 엣지(나뭇잎 등)에는 효과가 없습니다 — 그건 GL_SAMPLE_ALPHA_TO_COVERAGE나 FXAA가 필요합니다.

1-2. MSAA — 커스텀 FBO에 직접 (후처리와 함께)

HDR·블룸 같은 후처리를 하려면 오프스크린 FBO에 그려야 하는데, 그 FBO도 멀티샘플이어야 합니다. 멀티샘플 텍스처는 셰이더에서 바로 샘플링하기 번거로우므로, 멀티샘플 FBO에 그린 뒤 일반 FBO로 리졸브(blit)하는 2단계 패턴을 씁니다.

msaa_fbo.cppcpp
const int W = 1280, H = 720, SAMPLES = 4;

// ── (A) 멀티샘플 FBO: 여기에 씬을 그린다 ──
GLuint msFBO;  glGenFramebuffers(1, &msFBO);
glBindFramebuffer(GL_FRAMEBUFFER, msFBO);

// 멀티샘플 컬러 렌더버퍼
GLuint msColor; glGenRenderbuffers(1, &msColor);
glBindRenderbuffer(GL_RENDERBUFFER, msColor);
glRenderbufferStorageMultisample(GL_RENDERBUFFER, SAMPLES, GL_RGBA8, W, H);
glFramebufferRenderbuffer(GL_FRAMEBUFFER, GL_COLOR_ATTACHMENT0,
                          GL_RENDERBUFFER, msColor);

// 멀티샘플 깊이/스텐실 렌더버퍼
GLuint msDepth; glGenRenderbuffers(1, &msDepth);
glBindRenderbuffer(GL_RENDERBUFFER, msDepth);
glRenderbufferStorageMultisample(GL_RENDERBUFFER, SAMPLES, GL_DEPTH24_STENCIL8, W, H);
glFramebufferRenderbuffer(GL_FRAMEBUFFER, GL_DEPTH_STENCIL_ATTACHMENT,
                          GL_RENDERBUFFER, msDepth);

if (glCheckFramebufferStatus(GL_FRAMEBUFFER) != GL_FRAMEBUFFER_COMPLETE)
    std::fprintf(stderr, "멀티샘플 FBO 미완성\n");

// ── (B) 일반(단일샘플) FBO: 리졸브 결과를 담아 셰이더에서 샘플링 ──
GLuint resolveFBO; glGenFramebuffers(1, &resolveFBO);
glBindFramebuffer(GL_FRAMEBUFFER, resolveFBO);
GLuint screenTex; glGenTextures(1, &screenTex);
glBindTexture(GL_TEXTURE_2D, screenTex);
glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA8, W, H, 0, GL_RGBA, GL_UNSIGNED_BYTE, nullptr);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
glFramebufferTexture2D(GL_FRAMEBUFFER, GL_COLOR_ATTACHMENT0,
                       GL_TEXTURE_2D, screenTex, 0);

// ── 렌더 루프 ──
void renderFrame() {
    // 1) 멀티샘플 FBO에 씬 그리기
    glBindFramebuffer(GL_FRAMEBUFFER, msFBO);
    glEnable(GL_MULTISAMPLE);
    glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
    drawScene();

    // 2) 멀티샘플 → 단일샘플 리졸브 (blit)
    glBindFramebuffer(GL_READ_FRAMEBUFFER, msFBO);
    glBindFramebuffer(GL_DRAW_FRAMEBUFFER, resolveFBO);
    glBlitFramebuffer(0, 0, W, H, 0, 0, W, H,
                      GL_COLOR_BUFFER_BIT, GL_NEAREST);

    // 3) 이제 screenTex를 후처리 셰이더 입력으로 사용해 화면에 출력
    glBindFramebuffer(GL_FRAMEBUFFER, 0);
    glBindTexture(GL_TEXTURE_2D, screenTex);
    drawFullscreenQuad(postShader);
}
함정 glBlitFramebuffer로 리졸브할 때 소스와 대상의 크기가 같아야 GL_NEAREST가 유효합니다. 크기를 바꾸며 blit하면 멀티샘플 소스에서는 에러가 납니다. 또한 리졸브 대상은 반드시 단일샘플 FBO여야 합니다.

1-3. FXAA — 후처리 셰이더 (럭마 기반 엣지 블렌딩)

FXAA는 이미 렌더된 이미지를 입력으로 받아, 밝기(luma) 차이로 엣지를 찾고 그 방향으로 이웃 픽셀을 섞는 순수 화면공간 기법입니다. MSAA와 달리 알파 엣지·셰이더 앨리어싱까지 부드럽게 하지만, 약간 흐릿해집니다. 아래는 개념을 담은 간결한 자작 버전입니다.

fxaa.fragglsl
#version 330 core
in  vec2 vUV;
out vec4 FragColor;

uniform sampler2D uScene;     // 리졸브된(또는 후처리 전) 씬 컬러
uniform vec2      uTexel;     // 1.0 / 화면해상도

// RGB → 인지 밝기(luma)
float luma(vec3 c) { return dot(c, vec3(0.299, 0.587, 0.114)); }

void main() {
    // 중앙과 상하좌우 4방향의 밝기 샘플
    vec3  rgbM = texture(uScene, vUV).rgb;
    float lM = luma(rgbM);
    float lN = luma(texture(uScene, vUV + vec2(0.0, -uTexel.y)).rgb);
    float lS = luma(texture(uScene, vUV + vec2(0.0,  uTexel.y)).rgb);
    float lW = luma(texture(uScene, vUV + vec2(-uTexel.x, 0.0)).rgb);
    float lE = luma(texture(uScene, vUV + vec2( uTexel.x, 0.0)).rgb);

    // 국소 대비: 엣지가 아니면 그대로 통과 (성능 + 흐림 방지)
    float lMin = min(lM, min(min(lN, lS), min(lW, lE)));
    float lMax = max(lM, max(max(lN, lS), max(lW, lE)));
    float contrast = lMax - lMin;
    if (contrast < max(0.0312, lMax * 0.125)) {
        FragColor = vec4(rgbM, 1.0);
        return;
    }

    // 엣지 방향 추정: 수직/수평 밝기 기울기 비교
    float gradV = abs(lN + lS - 2.0 * lM);   // 수직 방향 변화
    float gradH = abs(lW + lE - 2.0 * lM);   // 수평 방향 변화
    bool horizontal = gradH >= gradV;
    vec2 dir = horizontal ? vec2(uTexel.x, 0.0) : vec2(0.0, uTexel.y);

    // 엣지를 가로질러 양쪽 이웃과 평균 → 계단 완화
    vec3 a = texture(uScene, vUV + dir).rgb;
    vec3 b = texture(uScene, vUV - dir).rgb;
    vec3 blended = (rgbM + a + b) / 3.0;

    // 대비가 클수록 강하게 섞는다
    float amount = clamp(contrast * 2.0, 0.0, 1.0);
    FragColor = vec4(mix(rgbM, blended, amount), 1.0);
}
주의 FXAA는 sRGB로 인코딩된(감마 적용된) 컬러 위에서 돌리는 것이 원칙입니다. luma가 인지 밝기 기준이라야 엣지가 잘 잡히기 때문입니다. 즉 톤매핑·감마 보정 이후의 이미지를 입력으로 넣으세요 (2절 참고).

1-4. TAA — 시간적 안티앨리어싱 (개념)

TAA는 매 프레임 카메라 투영에 서브픽셀 크기의 지터(jitter)를 주어 샘플 위치를 흩뿌린 뒤, 이전 프레임 결과를 현재 화면에 재투영(reprojection)해 지수 이동평균으로 누적합니다. 즉 "여러 프레임에 걸친 슈퍼샘플링"이라, 정지 화면에서는 MSAA보다 훨씬 매끈하고 셰이더 앨리어싱까지 잡습니다. 대신 (1) 지터 오프셋을 뷰-투영 행렬에 더하고, (2) 픽셀별 모션 벡터로 이전 프레임 위치를 찾아 히스토리 텍스처를 샘플링하며, (3) 이동/디스오클루전 시 유령상(ghosting)을 막기 위해 이웃 색상 범위로 히스토리를 클램프해야 합니다. 구현 난도가 높아 여기서는 개념만 짚고, 실무에서는 대개 엔진 제공 TAA를 튜닝해 사용합니다.

2. 감마 · 선형 색공간 (Gamma / Linear Space)

모니터는 색을 sRGB(≈ 감마 2.2) 곡선으로 표시하고, 이미지 파일(디퓨즈 텍스처 등)도 대부분 sRGB로 저장됩니다. 하지만 빛의 덧셈·곱셈(조명 계산)은 선형 공간에서 해야 물리적으로 옳습니다. 그래서 파이프라인은 항상 이 순서를 지킵니다:

  1. 읽을 때 디코드: sRGB 텍스처를 선형으로 변환해 읽는다.
  2. 선형에서 조명: 모든 라이팅·블렌딩을 선형 공간에서 수행한다.
  3. 출력할 때 인코드: 최종 색을 다시 sRGB로 변환해 화면에 쓴다.

2-1. 방법 A — 하드웨어에 맡기기 (권장)

디코드는 텍스처를 GL_SRGB8_ALPHA8 내부 포맷으로 만들면 GPU가 읽을 때 자동으로 선형화합니다. 인코드는 GL_FRAMEBUFFER_SRGB를 켜면 프레임버퍼에 쓸 때 자동으로 sRGB로 변환합니다. 셰이더 코드는 손댈 필요가 없습니다.

gamma_hw.cppcpp
// ── (1) 디코드: 컬러/디퓨즈 텍스처는 sRGB 내부 포맷으로 업로드 ──
//     주의: 노멀맵·러프니스·메탈릭 등 "데이터" 텍스처는 GL_RGBA8(선형)로!
GLuint loadColorTexture(const unsigned char* pixels, int w, int h) {
    GLuint tex; glGenTextures(1, &tex);
    glBindTexture(GL_TEXTURE_2D, tex);
    glTexImage2D(GL_TEXTURE_2D, 0,
                 GL_SRGB8_ALPHA8,          // ← 하드웨어가 읽을 때 선형으로 디코드
                 w, h, 0, GL_RGBA, GL_UNSIGNED_BYTE, pixels);
    glGenerateMipmap(GL_TEXTURE_2D);
    return tex;
}

// ── (2) 인코드: 최종 출력 시 sRGB로 자동 변환 ──
glEnable(GL_FRAMEBUFFER_SRGB);   // 기본 프레임버퍼에 쓸 때 선형→sRGB 인코드
// 이제 셰이더는 전부 선형 색으로 계산하면 된다. 감마 코드가 필요 없다.

2-2. 방법 B — 셰이더에서 수동으로

SRGB 프레임버퍼를 쓸 수 없는 환경(일부 후처리 체인, WebGL 구버전 등)에서는 셰이더 끝에서 직접 pow(color, 1/2.2)로 인코드합니다. 텍스처를 선형 포맷으로 읽었다면 셰이더에서 pow(texColor, 2.2)로 디코드도 직접 해야 합니다.

gamma_manual.fragglsl
#version 330 core
in  vec2 vUV;
out vec4 FragColor;
uniform sampler2D uAlbedo;     // GL_RGBA8(선형 포맷)로 올린 sRGB 이미지

const float GAMMA = 2.2;

void main() {
    // (1) 디코드: sRGB → 선형 (하드웨어 sRGB 텍스처를 안 썼을 때만)
    vec3 albedo = pow(texture(uAlbedo, vUV).rgb, vec3(GAMMA));

    // (2) 선형 공간에서 조명 계산
    vec3 color = doLighting(albedo);   // 모든 add/mul은 선형에서

    // (3) 인코드: 선형 → sRGB (화면 출력 직전 한 번만)
    color = pow(color, vec3(1.0 / GAMMA));
    FragColor = vec4(color, 1.0);
}
함정 이중 보정 금지. GL_FRAMEBUFFER_SRGB를 켠 상태에서 셰이더에서도 pow(color, 1/2.2)를 하면 감마가 두 번 적용돼 화면이 지나치게 밝고 뿌예집니다. 마찬가지로 GL_SRGB8_ALPHA8 텍스처를 쓰면서 셰이더에서 또 pow(.., 2.2)로 디코드하면 색이 과도하게 어두워집니다. 하드웨어 방식과 수동 방식은 절대 섞지 마세요.

3. 밉맵 · 텍스처 필터링 (Mipmapping / Filtering)

텍스처가 화면에서 작게(멀리) 그려지면 한 픽셀이 텍스처의 여러 텍셀에 걸칩니다. 이때 한 텍셀만 찍으면 프레임마다 다른 텍셀이 뽑혀 지글거리는 축소 앨리어싱(minification aliasing)이 생깁니다. 밉맵은 텍스처를 절반씩 미리 축소한 피라미드를 만들어, 화면 크기에 맞는 레벨을 골라 샘플링함으로써 이 문제를 해결합니다.

mipmap.cppcpp
GLuint tex; glGenTextures(1, &tex);
glBindTexture(GL_TEXTURE_2D, tex);
glTexImage2D(GL_TEXTURE_2D, 0, GL_SRGB8_ALPHA8, w, h, 0,
             GL_RGBA, GL_UNSIGNED_BYTE, pixels);

// ── 밉맵 자동 생성 (base 레벨로부터 축소 피라미드 구성) ──
glGenerateMipmap(GL_TEXTURE_2D);

// ── 축소 필터: 트라이리니어 (레벨 내 + 레벨 간 이중 보간) ──
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR_MIPMAP_LINEAR);
//   대안 비교:
//   GL_NEAREST_MIPMAP_NEAREST → 가장 싸지만 레벨 전환선이 보임
//   GL_LINEAR_MIPMAP_NEAREST  → 바이리니어 + 레벨 하드컷 (구형 게임 룩)
//   GL_LINEAR_MIPMAP_LINEAR   → 트라이리니어, 표준 선택

// ── 확대 필터: 밉맵과 무관 (한 레벨만 쓰므로) ──
glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
//   GL_NEAREST 로 두면 픽셀아트 느낌의 또렷한 확대

3-1. 이방성 필터링 (Anisotropic Filtering)

바닥처럼 카메라에 대해 비스듬한 표면은 한 픽셀이 한 방향으로 길게 늘어난 텍셀 영역을 덮습니다. 트라이리니어는 등방성(정사각) 샘플이라 이런 경사면을 과도하게 흐리는데, 이방성 필터링은 늘어난 방향으로 더 많이 샘플링해 원거리 바닥의 선명함을 살립니다. OpenGL 4.6 코어이거나 GL_EXT_texture_filter_anisotropic 확장이 있으면 씁니다.

anisotropic.cppcpp
// GL 4.6 코어에서는 GL_TEXTURE_MAX_ANISOTROPY 가 표준 토큰.
// (확장만 있을 경우 GL_TEXTURE_MAX_ANISOTROPY_EXT / _MAX_.._EXT 를 사용)
#ifndef GL_TEXTURE_MAX_ANISOTROPY
#define GL_TEXTURE_MAX_ANISOTROPY         0x84FE
#define GL_MAX_TEXTURE_MAX_ANISOTROPY     0x84FF
#endif

bool hasAnisotropy() {
    // GL 4.6 코어이거나 확장 문자열이 있으면 지원
    GLint major = 0, minor = 0;
    glGetIntegerv(GL_MAJOR_VERSION, &major);
    glGetIntegerv(GL_MINOR_VERSION, &minor);
    if (major > 4 || (major == 4 && minor >= 6)) return true;
    // (확장 조회는 glGetStringi 로 GL_EXTENSIONS 순회)
    return false;
}

void enableAnisotropy(GLuint tex) {
    if (!hasAnisotropy()) return;
    GLfloat maxAniso = 1.0f;
    glGetFloatv(GL_MAX_TEXTURE_MAX_ANISOTROPY, &maxAniso);  // 보통 16
    glBindTexture(GL_TEXTURE_2D, tex);
    // 하드웨어 최대치와 원하는 값(예: 8x) 중 작은 쪽
    glTexParameterf(GL_TEXTURE_2D, GL_TEXTURE_MAX_ANISOTROPY,
                    (maxAniso < 8.0f) ? maxAniso : 8.0f);
}
성능 이방성은 샘플 수를 늘리므로 텍스처 대역폭을 먹습니다. 16x가 늘 8x보다 눈에 띄게 좋은 건 아니니 8x 정도가 화질/비용 균형점인 경우가 많습니다. 밉맵이 없으면 이방성도 의미가 없으니 반드시 glGenerateMipmap + GL_LINEAR_MIPMAP_LINEAR와 함께 쓰세요.

4. 투명과 렌더링 순서 (Transparency & Ordering)

유리·물·연기처럼 뒤가 비치는 물체는 알파 블렌딩으로 그립니다. 문제는 블렌딩이 순서에 의존한다는 점입니다. 뒤 물체가 먼저 프레임버퍼에 있어야 앞 물체가 그 위에 섞이기 때문입니다. 그래서 표준 전략은 2패스 + 정렬입니다.

blend_setup.cppcpp
glEnable(GL_BLEND);
// 결과 = src.rgb * src.a + dst.rgb * (1 - src.a)  → 표준 알파 오버
glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);

4-1. 왜 정렬이 필요한가

깊이 테스트는 "더 먼 픽셀을 버리는" 방식이라, 반투명 물체를 앞에서부터 그리면 뒤 물체가 깊이 테스트에 걸려 아예 그려지지 않거나, 그려지더라도 이미 위에 있는 색과 섞을 대상이 없습니다. 따라서 반투명은 반드시 카메라에서 먼 것부터 가까운 것 순(back-to-front)으로 그려야 색이 올바르게 누적됩니다.

4-2. 2패스 렌더링

  1. 불투명 패스: 깊이 쓰기 ON, 블렌딩 OFF. 순서 무관하게 다 그린다.
  2. 반투명 물체를 카메라 거리로 정렬(먼 것 먼저)한다.
  3. 반투명 패스: 깊이 테스트는 유지하되 깊이 쓰기는 끄고 (glDepthMask(GL_FALSE)) 블렌딩 ON으로 그린다.
two_pass_transparency.cppcpp
#include <algorithm>
#include <vector>
#include <glm/glm.hpp>

struct Renderable {
    glm::vec3 position;
    bool      transparent;
    // ... 메시 핸들, 머티리얼 등
};

void renderScene(std::vector<Renderable>& objects, const glm::vec3& camPos) {
    // ── 패스 1: 불투명 (깊이 쓰기 ON, 블렌딩 OFF) ──
    glEnable(GL_DEPTH_TEST);
    glDepthMask(GL_TRUE);
    glDisable(GL_BLEND);
    for (auto& o : objects)
        if (!o.transparent) drawObject(o);

    // ── 반투명만 골라 카메라 거리 제곱으로 정렬 (먼 것 → 가까운 것) ──
    std::vector<Renderable*> blended;
    for (auto& o : objects)
        if (o.transparent) blended.push_back(&o);

    std::sort(blended.begin(), blended.end(),
        [&](const Renderable* a, const Renderable* b) {
            float da = glm::dot(a->position - camPos, a->position - camPos);
            float db = glm::dot(b->position - camPos, b->position - camPos);
            return da > db;   // 거리 큰 것이 앞에 오도록 (먼저 그림)
        });

    // ── 패스 2: 반투명 (깊이 쓰기 OFF, 블렌딩 ON) ──
    glEnable(GL_BLEND);
    glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
    glDepthMask(GL_FALSE);          // 반투명끼리 깊이로 서로 가리지 않게
    for (Renderable* o : blended)
        drawObject(*o);
    glDepthMask(GL_TRUE);           // 다음 프레임 위해 원상복구
}
함정 반투명 패스에서 glDepthMask(GL_FALSE)로 쓰기만 끄고 GL_DEPTH_TEST 자체는 켜둬야 합니다. 테스트를 끄면 불투명 물체 뒤에 있어야 할 유리가 앞에 그려집니다. 루프가 끝나면 반드시 glDepthMask(GL_TRUE)로 되돌리세요 — 안 그러면 다음 프레임의 불투명 물체가 깊이를 못 써서 깨집니다.
주의 정렬은 오브젝트 단위라 서로 관통하는 반투명 메시나 자기 자신이 겹치는 메시에서는 여전히 틀립니다. 이를 순서 독립적으로 푸는 것이 OIT(Order-Independent Transparency)입니다. 실무에서 흔한 Weighted Blended OIT는 색과 가중치를 누적한 뒤 나눠 근사하므로 정렬이 필요 없습니다(품질은 근사). 대량의 파티클·식생에 유용합니다.

5. 앰비언트 오클루전 (Ambient Occlusion)

AO는 틈새·모서리·접촉면처럼 주변에 가려져 간접광이 덜 도달하는 부분을 어둡게 만들어 형태와 접지감을 살립니다. 가장 널리 쓰이는 SSAO는 화면공간 기법으로, G-버퍼(깊이·노멀)만으로 오클루전을 추정합니다. 전체 구현은 6장에 있으므로 여기서는 요약과 대안만 정리합니다 → 6장 · 화면공간 기법(SSAO).

  1. 입력: 뷰공간 깊이·노멀(디퍼드라면 G-버퍼에서 바로).
  2. 오클루전: 각 픽셀 주변 반구에 커널 샘플을 뿌려, 이웃 깊이가 더 앞이면 "가려짐"으로 카운트.
  3. 블러: 노이즈 회전으로 생긴 지글거림을 좁은 박스 블러로 완화.
  4. 합성: AO 값을 최종 조명의 앰비언트 항에만 곱한다(직접광엔 곱하지 않음).

대안도 알아두면 좋습니다. HBAO(Horizon-Based)는 깊이 지평선 각도로 더 정확하게 추정하고, GTAO(Ground Truth AO)는 물리적으로 더 정합한 최신 표준입니다. 실시간 계산이 부담되거나 정적 지오메트리라면 오프라인에서 구워둔 베이크드 AO 맵을 텍스처로 곱하는 방식이 여전히 가장 저렴하고 안정적입니다.

성능 SSAO 커널은 픽셀당 샘플 수(예: 16~64)에 비용이 비례합니다. 커널을 저해상도(1/2)로 계산한 뒤 업스케일하고, 반드시 블러 패스를 붙이는 것이 실무의 표준 절충입니다.

6. LOD · 컬링 (Level of Detail & Culling)

대규모 씬의 핵심은 "안 보이는 건 그리지 않고, 멀리 있는 건 대충 그린다"입니다. 프러스텀 컬링으로 시야 밖 물체를 버리고, 거리 기반 LOD로 먼 물체는 저폴리곤 메시를 쓰며, 같은 물체가 많으면 인스턴싱으로 드로우콜을 뭉칩니다.

6-1. 프러스텀 평면 6개 추출

뷰-투영 행렬 하나에서 절두체의 6개 평면(좌·우·하·상·근·원)을 뽑아낼 수 있습니다. 각 평면은 ax + by + cz + d = 0의 (a,b,c,d)이며, 행렬의 행을 더하고 빼서 얻습니다.

frustum.hppcpp
#include <glm/glm.hpp>
#include <array>

struct Plane {
    glm::vec3 n;   // 정규화된 법선 (프러스텀 안쪽을 향함)
    float     d;   // 부호 있는 원점 거리
    // 점 p의 부호 있는 거리: 양수면 평면 안쪽
    float distance(const glm::vec3& p) const { return glm::dot(n, p) + d; }
};

// 뷰-투영 행렬 m = projection * view 에서 6평면 추출 (Gribb–Hartmann 방식)
std::array<Plane, 6> extractFrustumPlanes(const glm::mat4& m) {
    std::array<Plane, 6> planes;
    // glm은 열우선이라 m[col][row]. 행벡터를 r0..r3로 꺼내둔다.
    glm::vec4 r0(m[0][0], m[1][0], m[2][0], m[3][0]);
    glm::vec4 r1(m[0][1], m[1][1], m[2][1], m[3][1]);
    glm::vec4 r2(m[0][2], m[1][2], m[2][2], m[3][2]);
    glm::vec4 r3(m[0][3], m[1][3], m[2][3], m[3][3]);

    glm::vec4 raw[6] = {
        r3 + r0,   // 좌
        r3 - r0,   // 우
        r3 + r1,   // 하
        r3 - r1,   // 상
        r3 + r2,   // 근 (near)
        r3 - r2    // 원 (far)
    };
    for (int i = 0; i < 6; ++i) {
        glm::vec3 n(raw[i].x, raw[i].y, raw[i].z);
        float len = glm::length(n);          // 정규화: 거리 스케일을 맞춤
        planes[i].n = n / len;
        planes[i].d = raw[i].w / len;
    }
    return planes;
}

6-2. AABB vs 프러스텀 테스트

물체를 축 정렬 바운딩 박스(AABB)로 감싸고, 6개 평면 각각에 대해 "박스 전체가 평면 바깥에 있는가"를 검사합니다. 하나라도 완전히 바깥이면 보이지 않으므로 컬링합니다. 평면의 법선 방향에서 가장 먼 꼭짓점(positive vertex)만 검사하면 충분합니다.

frustum_cull.cppcpp
struct AABB { glm::vec3 min, max; };

// 프러스텀 안(또는 걸침)에 있으면 true → 그려야 함
bool aabbInFrustum(const std::array<Plane, 6>& planes, const AABB& box) {
    for (const Plane& pl : planes) {
        // 법선 방향으로 가장 "안쪽에 가까운" 꼭짓점 = positive vertex
        glm::vec3 pv;
        pv.x = (pl.n.x >= 0.0f) ? box.max.x : box.min.x;
        pv.y = (pl.n.y >= 0.0f) ? box.max.y : box.min.y;
        pv.z = (pl.n.z >= 0.0f) ? box.max.z : box.min.z;

        // 가장 유리한 꼭짓점조차 평면 바깥이면 박스 전체가 바깥 → 컬링
        if (pl.distance(pv) < 0.0f)
            return false;
    }
    return true;   // 어떤 평면에도 완전히 밀려나지 않음 → 보임
}

6-3. 거리 기반 LOD 선택

카메라와의 거리로 어느 해상도의 메시를 그릴지 고릅니다. 임계값은 화면 픽셀 크기·성능 목표에 맞춰 튜닝합니다. 경계에서의 "팝핑"이 거슬리면 두 LOD를 짧게 크로스페이드하거나 디더링합니다.

lod_pick.cppcpp
struct Model {
    Mesh lod0, lod1, lod2;   // 고 → 중 → 저 폴리곤
    glm::vec3 center;
};

Mesh& pickLOD(Model& model, const glm::vec3& camPos) {
    // sqrt를 피하려 거리 제곱으로 비교 (임계값도 제곱해 둠)
    float d2 = glm::dot(model.center - camPos, model.center - camPos);
    const float near2 = 15.0f * 15.0f;   // 15m 이내: 최고 디테일
    const float mid2  = 50.0f * 50.0f;   // 50m 이내: 중간
    if (d2 < near2) return model.lod0;
    if (d2 < mid2)  return model.lod1;
    return model.lod2;                    // 그 밖: 최저 (또는 임포스터)
}

// 사용: 컬링 통과한 것만 LOD 골라 그린다
void drawModels(std::vector<Model>& models,
                const std::array<Plane,6>& frustum,
                const glm::vec3& camPos) {
    for (Model& m : models) {
        AABB box = boundsOf(m);
        if (!aabbInFrustum(frustum, box)) continue;   // 시야 밖 스킵
        draw(pickLOD(m, camPos));
    }
}

6-4. 인스턴싱 (같은 메시 대량 렌더)

풀·바위·군중처럼 같은 메시를 수천 개 그릴 때, 드로우콜을 하나로 묶는 것이 인스턴싱입니다. 인스턴스별 변환 행렬을 VBO에 담아 glVertexAttribDivisor로 "인스턴스마다 한 번 전진"하게 설정한 뒤 glDrawElementsInstanced로 한 번에 그립니다.

instancing.cppcpp
// vao 는 메시(위치/노멀/UV)가 이미 바인딩된 상태라고 가정.
// 인스턴스별 model 행렬 배열을 준비한다.
std::vector<glm::mat4> models = buildInstanceMatrices();  // 예: 5000개
const int count = (int)models.size();

// ── 인스턴스 행렬용 VBO ──
GLuint instanceVBO; glGenBuffers(1, &instanceVBO);
glBindVertexArray(vao);
glBindBuffer(GL_ARRAY_BUFFER, instanceVBO);
glBufferData(GL_ARRAY_BUFFER, count * sizeof(glm::mat4),
             models.data(), GL_STATIC_DRAW);

// mat4 = vec4 4개 → 어트리뷰트 슬롯 4칸(loc 3,4,5,6) 사용
const std::size_t vec4Size = sizeof(glm::vec4);
for (int i = 0; i < 4; ++i) {
    glEnableVertexAttribArray(3 + i);
    glVertexAttribPointer(3 + i, 4, GL_FLOAT, GL_FALSE,
                          sizeof(glm::mat4), (void*)(i * vec4Size));
    glVertexAttribDivisor(3 + i, 1);   // ← 정점마다가 아니라 "인스턴스마다" 전진
}

// ── 렌더: 드로우콜 한 번으로 count개 ──
glBindVertexArray(vao);
glDrawElementsInstanced(GL_TRIANGLES, indexCount, GL_UNSIGNED_INT, 0, count);
성능 인스턴싱의 이득은 드로우콜(CPU→GPU 오버헤드) 절감이지 GPU 정점 처리량 자체를 줄이진 않습니다. 그래서 프러스텀 컬링·LOD와 함께 써야 진짜로 빨라집니다. 인스턴스 데이터가 매 프레임 바뀌면 GL_DYNAMIC_DRAW로 만들고, CPU에서 프러스텀 컬링해 보이는 인스턴스만 버퍼에 채워 넣으세요.
